cmake_minimum_required(VERSION 3.24)
project(PartyLight VERSION 0.4.3 LANGUAGES CXX)
set(PARTYLIGHT_RUNTIME "flatrim" CACHE STRING "Target runtime: flatrim or vr")
set_property(CACHE PARTYLIGHT_RUNTIME PROPERTY STRINGS flatrim vr)
set(COMMONLIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/CommonLibSSE-NG" CACHE PATH "alandtse CommonLibSSE-NG checkout")
set(COMMONLIB_LIBRARY "" CACHE FILEPATH "Optional matching SE+AE /MD release CommonLib library")
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/vcpkg_installed/x64-windows-static-md" CACHE PATH "Static MD dependencies")
list(APPEND CMAKE_PREFIX_PATH "${DEPS}")
set(CMAKE_MAP_IMPORTED_CONFIG_RELWITHDEBINFO Release)
find_package(spdlog CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(directxtk CONFIG REQUIRED)
if(NOT PARTYLIGHT_RUNTIME MATCHES "^(flatrim|vr)$")
    message(FATAL_ERROR "PARTYLIGHT_RUNTIME must be flatrim or vr")
endif()
if(PARTYLIGHT_RUNTIME STREQUAL "vr" OR NOT EXISTS "${COMMONLIB_LIBRARY}")
    if(PARTYLIGHT_RUNTIME STREQUAL "vr")
        set(ENABLE_SKYRIM_VR ON CACHE BOOL "" FORCE)
        set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
        set(ENABLE_SKYRIM_AE OFF CACHE BOOL "" FORCE)
    else()
        set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
        set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
        set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
    endif()
    set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
    set(RAPIDCSV_INCLUDE_DIRS "${DEPS}/include" CACHE PATH "rapidcsv headers")
    add_subdirectory("${COMMONLIB_DIR}" commonlib)
elseif(PARTYLIGHT_RUNTIME STREQUAL "flatrim")
    add_library(CommonLibSSE::CommonLibSSE STATIC IMPORTED)
    set_target_properties(CommonLibSSE::CommonLibSSE PROPERTIES IMPORTED_CONFIGURATIONS Release IMPORTED_LOCATION_RELEASE "${COMMONLIB_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${COMMONLIB_DIR}/include;${COMMONLIB_DIR}/extern/openvr/headers" INTERFACE_COMPILE_DEFINITIONS "ENABLE_SKYRIM_AE=1;ENABLE_SKYRIM_SE=1;HAS_SKYRIM_MULTI_TARGETING=1")
endif()
include("${COMMONLIB_DIR}/cmake/CommonLibSSE.cmake")
add_commonlibsse_plugin(PartyLight AUTHOR "nosimo" SOURCES src/plugin.cpp src/SurvivalBridge.cpp src/CanineBridge.cpp src/OverheadBridge.cpp)
target_compile_features(PartyLight PRIVATE cxx_std_23)
target_compile_definitions(PartyLight PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(PartyLight PRIVATE /Zc:preprocessor /permissive- /EHsc /utf-8)
target_precompile_headers(PartyLight PRIVATE src/PCH.h)
target_include_directories(PartyLight PRIVATE include)
target_link_libraries(PartyLight PRIVATE spdlog::spdlog nlohmann_json::nlohmann_json Microsoft::DirectXTK advapi32 bcrypt d3d11 d3dcompiler dbghelp dxgi ole32 version crypt32)
include(CTest)
if(BUILD_TESTING)
    add_executable(PartyLightLifecycleTest tests/lifecycle.cpp)
    target_include_directories(PartyLightLifecycleTest PRIVATE src)
    target_compile_features(PartyLightLifecycleTest PRIVATE cxx_std_23)
    target_compile_options(PartyLightLifecycleTest PRIVATE /EHsc)
    add_test(NAME PartyLightLifecycle COMMAND PartyLightLifecycleTest)
    add_executable(PartyLightFollowerTest tests/followers.cpp)
    target_include_directories(PartyLightFollowerTest PRIVATE src)
    target_compile_features(PartyLightFollowerTest PRIVATE cxx_std_23)
    target_compile_options(PartyLightFollowerTest PRIVATE /EHsc)
    add_test(NAME PartyLightFollowers COMMAND PartyLightFollowerTest)
    add_executable(PartyLightCanineTest tests/canine.cpp)
    target_include_directories(PartyLightCanineTest PRIVATE src include)
    target_compile_features(PartyLightCanineTest PRIVATE cxx_std_23)
    target_compile_options(PartyLightCanineTest PRIVATE /EHsc)
    add_test(NAME PartyLightCanine COMMAND PartyLightCanineTest)
endif()
